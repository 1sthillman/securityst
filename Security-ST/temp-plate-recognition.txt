  function loadScriptWithTimeout(src, timeoutMs){
    return new Promise(function(resolve, reject){
      var done = false;
      var s = document.createElement('script');
      s.src = src; s.async = true;
      var t = setTimeout(function(){ if(!done){ done=true; reject(new Error('timeout: '+src)); } }, timeoutMs || 9000);
      s.onload = function(){ if(!done){ done=true; clearTimeout(t); resolve(); } };
      s.onerror = function(){ if(!done){ done=true; clearTimeout(t); reject(new Error('yÃ¼klenemedi: '+src)); } };
      document.head.appendChild(s);
    });
  }

  function beginLazyEngineLoad(){
    if(engineLoadStarted) return;
    engineLoadStarted = true;

  /* =========================================================
     5. Ã–N Ä°ÅLEME ZÄ°NCÄ°RÄ° (her adÄ±m hata verirse bir Ã¶ncekiyle devam eder)
  ========================================================= */
  // OpenCV tabanlÄ± zincir: gri tonlama -> parlama bastÄ±rma -> gÃ¼rÃ¼ltÃ¼ azaltma
  // -> CLAHE kontrast -> adaptif eÅŸikleme -> eÄŸim dÃ¼zeltme (kontur + dÃ¶ndÃ¼rme)
  function preprocessWithOpenCV(canvas){
    var stageCanvas = canvas;
    try{
      var src = cv.imread(canvas);
      var work = src;

      try{ var gray = new cv.Mat(); cv.cvtColor(work, gray, cv.COLOR_RGBA2GRAY); work.delete(); work = gray; }catch(e){}

      try{
        var mask = new cv.Mat();
        cv.threshold(work, mask, 235, 255, cv.THRESH_BINARY);
        var kernel = cv.Mat.ones(3,3, cv.CV_8U);
        cv.dilate(mask, mask, kernel);
        var inpainted = new cv.Mat();
        cv.inpaint(work, mask, inpainted, 4, cv.INPAINT_TELEA);
        work.delete(); mask.delete(); kernel.delete();
        work = inpainted;
      }catch(e){ /* parlama bastÄ±rma baÅŸarÄ±sÄ±z oldu, atlanÄ±yor */ }

      try{ var blurred = new cv.Mat(); cv.medianBlur(work, blurred, 3); work.delete(); work = blurred; }catch(e){}

      try{
        var clahe = new cv.CLAHE(2.0, new cv.Size(8,8));
        var claheOut = new cv.Mat();
        clahe.apply(work, claheOut);
        work.delete(); work = claheOut;
      }catch(e){ /* CLAHE baÅŸarÄ±sÄ±z, ham gri ton ile devam */ }

      // eÄŸim dÃ¼zeltme: en bÃ¼yÃ¼k konturu bul, minAreaRect aÃ§Ä±sÄ±na gÃ¶re dÃ¶ndÃ¼r
      try{
        var edges = new cv.Mat();
        cv.threshold(work, edges, 0, 255, cv.THRESH_BINARY+cv.THRESH_OTSU);
        var contours = new cv.MatVector();
        var hierarchy = new cv.Mat();
        cv.findContours(edges, contours, hierarchy, cv.RETR_EXTERNAL, cv.CHAIN_APPROX_SIMPLE);
        var maxArea = 0, maxIdx = -1;
        for(var i=0;i<contours.size();i++){
          var a = cv.contourArea(contours.get(i));
          if(a > maxArea){ maxArea = a; maxIdx = i; }
        }
        if(maxIdx >= 0 && maxArea > work.cols*work.rows*0.15){
          var rect = cv.minAreaRect(contours.get(maxIdx));
          var angle = rect.angle;
          if(angle < -45) angle += 90;
          if(Math.abs(angle) > 1 && Math.abs(angle) <= 15){
            var center = new cv.Point(work.cols/2, work.rows/2);
            var M = cv.getRotationMatrix2D(center, angle, 1);
            var rotated = new cv.Mat();
            cv.warpAffine(work, rotated, M, new cv.Size(work.cols, work.rows), cv.INTER_LINEAR, cv.BORDER_REPLICATE);
            work.delete(); M.delete(); work = rotated;
          }
        }
        edges.delete(); contours.delete(); hierarchy.delete();
      }catch(e){ /* eÄŸim dÃ¼zeltme baÅŸarÄ±sÄ±z, dÃ¼zeltilmemiÅŸ gÃ¶rÃ¼ntÃ¼yle devam */ }

      try{
        var thresh = new cv.Mat();
        cv.adaptiveThreshold(work, thresh, 255, cv.ADAPTIVE_THRESH_GAUSSIAN_C, cv.THRESH_BINARY, 31, 15);
        work.delete(); work = thresh;
      }catch(e){ /* adaptif eÅŸikleme baÅŸarÄ±sÄ±z, kontrast-artÄ±rÄ±lmÄ±ÅŸ gri ton ile devam */ }

      var out = document.createElement('canvas');
      out.width = work.cols; out.height = work.rows;
      cv.imshow(out, work);
      work.delete();
      return out;
    }catch(e){
      return stageCanvas; // OpenCV tamamen baÅŸarÄ±sÄ±z oldu: ham kareyle devam
    }
  }

  // Daha hafif OpenCV varyantÄ±: adaptif eÅŸikleme ve eÄŸim dÃ¼zeltme olmadan, sadece
  // gri tonlama + parlama bastÄ±rma + kontrast artÄ±rma. BazÄ± plakalar ikili (binary)
  // gÃ¶rÃ¼ntÃ¼den Ã§ok, kontrastÄ± artÄ±rÄ±lmÄ±ÅŸ gri tondan daha iyi okunur.

  // OpenCV yoksa devreye giren basit yedek: gri tonlama + sabit eÅŸik (canvas pikselleri Ã¼zerinden)
  function preprocessBasic(canvas){
    try{
      var ctx = canvas.getContext('2d');
      var img = ctx.getImageData(0,0,canvas.width,canvas.height);
      var d = img.data;
      for(var i=0;i<d.length;i+=4){
        var v = 0.299*d[i] + 0.587*d[i+1] + 0.114*d[i+2];
        v = v > 140 ? 255 : (v < 90 ? 0 : v);
        d[i]=d[i+1]=d[i+2]=v;
      }
      ctx.putImageData(img,0,0);
      return canvas;
    }catch(e){ return canvas; }
  }

  var tesseractWorkerPromise = null;
  function getWorker(){
    if(tesseractWorkerPromise) return tesseractWorkerPromise;
    tesseractWorkerPromise = Tesseract.createWorker('eng', 1, { logger: function(){} }).then(function(worker){
      return worker.setParameters({ tessedit_char_whitelist: 'ABCDEFGHIJKLMNOPRSTUVYZ0123456789 ' }).then(function(){ return worker; });
    });
    return tesseractWorkerPromise;
  }

  function ocrCanvas(canvas){
    return getWorker().then(function(worker){
      return worker.recognize(canvas);
    }).then(function(res){
      return { text: normalizePlate(res.data.text), conf: res.data.confidence || 0 };
    });
  }

  function ocrFrame(processedCanvas){
    if(vehicleMode !== 'moto'){
      return ocrCanvas(processedCanvas);
    }
    // motosiklet: kÄ±rpÄ±lan alanÄ± yatay olarak ikiye bÃ¶l, ayrÄ± ayrÄ± oku, birleÅŸtir
    var w = processedCanvas.width, h = processedCanvas.height;
    var topC = document.createElement('canvas'); topC.width=w; topC.height=Math.round(h*0.55);
    topC.getContext('2d').drawImage(processedCanvas, 0,0,w,topC.height, 0,0,w,topC.height);
    var botC = document.createElement('canvas'); botC.width=w; botC.height=h-topC.height;
    botC.getContext('2d').drawImage(processedCanvas, 0,topC.height,w,botC.height, 0,0,w,botC.height);
    return Promise.all([ocrCanvas(topC), ocrCanvas(botC)]).then(function(r){
      return { text: (r[0].text+' '+r[1].text).replace(/\s+/g,' ').trim(), conf: (r[0].conf+r[1].conf)/2 };
    });
  }

  var isCapturing = false;

  function runCapture(){
    if(isCapturing) return;

  function runOcrOnVariants(variants){
    var results = [];
    var i = 0;
    function next(){
      if(i >= variants.length) return finishVoting(results);
      return ocrFrame(variants[i].canvas).then(function(r){
        r.label = variants[i].label; results.push(r); i++; return next();
      }).catch(function(){
        results.push({ text:'', conf:0, label: variants[i].label }); i++; return next();
      });
    }
    return next().catch(function(){
      showCamMsg('FotoÄŸraf iÅŸlenirken bir sorun oluÅŸtu. Tekrar deneyin ya da elle girin.');
      isCapturing = false; $('captureBtn').disabled = false;
    });
  }

  function finishVoting(results){
    isCapturing = false;
    $('captureBtn').disabled = false;

    var counts = {};
    results.forEach(function(r){
      if(!r.text) return;
      counts[r.text] = (counts[r.text]||0) + 1;
    });
    var bestText = '', bestCount = 0;
    Object.keys(counts).forEach(function(t){ if(counts[t] > bestCount){ bestCount = counts[t]; bestText = t; } });

    var chosen, confidenceLevel;
    if(bestCount >= 2){
      chosen = bestText; confidenceLevel = 'high';
    } else {
      var top = results.slice().sort(function(a,b){ return b.conf-a.conf; })[0] || {text:'', conf:0};
      chosen = top.text; confidenceLevel = chosen ? 'low' : 'none';
    }

    $('frameDetail').textContent = results.length > 1
      ? results.map(function(r){ return r.label+': "'+(r.text||'â€”')+'" ('+Math.round(r.conf)+'%)'; }).join('  â€¢  ')
      : '';

    showSuggestion(chosen, confidenceLevel);
  }
